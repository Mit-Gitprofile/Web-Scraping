How It Works

Four clean steps from URL to structured CSV dataset.

01

Loop Through Pages

Iterates over 10 paginated search result pages for the query "mobile" on Flipkart using the ?page=N URL parameter.

02

Fetch & Parse HTML

Sends HTTP GET requests via requests and parses the response with BeautifulSoup using the html.parser engine.

03

Extract Data Fields

Targets specific CSS class selectors to extract product name, price, specifications description, and customer review ratings.

04

Export to CSV

Assembles all collected lists into a pandas DataFrame and exports it as a structured data.csv file.

import pandas as pd
import requests
from bs4 import BeautifulSoup

# ── Initialize storage lists ──────────────────────────────────
product_name = []
prices       = []
description  = []
reviews      = []

# ── Paginated scraping loop (10 pages) ───────────────────────
for i in range(0, 10):

    url = (
        "https://www.flipkart.com/search?q=mobile"
        "&otracker=search&marketplace=FLIPKART"
        "&as-show=on&as=off&page=" + str(i)
    )

    # Send HTTP GET request
    r = requests.get(url)

    # Parse HTML response
    soup = BeautifulSoup(r.text, "html.parser")

    # Locate the main product grid container
    box = soup.find("div", class_="DOjaWF gdgoEp")

    # ── Extract: Product Names ───────────────────────────────────
    names = box.find_all("div", class_="KzDlHZ")
    for item in names:
        product_name.append(item.text)

    # ── Extract: Prices ──────────────────────────────────────────
    price = box.find_all("div", class_="Nx9bqj _4b5DiR")
    for item in price:
        prices.append(item.text)

    # ── Extract: Descriptions / Specs ────────────────────────────
    deccript = box.find_all("ul", class_="G4BRas")
    for item in deccript:
        description.append(item.text)

    # ── Extract: Review Ratings ──────────────────────────────────
    review = box.find_all("div", class_="XQDdHH")
    for item in review:
        reviews.append(item.text)

# ── Build DataFrame & Export ──────────────────────────────────
dataframe = pd.DataFrame({
    "Name":        product_name,
    "Price":       prices,
    "Description": description,
    "Review":      reviews
})

dataframe.to_csv("data.csv")
print(f"✅ Scraped {len(dataframe)} records successfully!")
